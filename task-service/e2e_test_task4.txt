# 1. Пересобрать worker
cd ~/task-manager-microservices/task-service
docker compose build task-worker
docker compose up -d task-worker

# 2. Ждать, пока в логах появится подключение к RabbitMQ
sleep 5
docker logs task-worker --tail 20 | grep -i "rabbit\|startup\|connected"

# 3. Новая задача
RESPONSE=$(curl -s -X POST http://localhost:8000/api/v1/tasks \
  -H "Content-Type: application/json" -H "X-User-Name: testuser" \
  -d '{"title": "E2E escalation v4", "priority": "low", "assignee": "dev@example.com"}')
TASK_ID=$(echo "$RESPONSE" | python3 -c "import sys,json; print(json.load(sys.stdin)['id'])")
echo "TASK_ID=$TASK_ID"

# 4. Сдвинуть дату
docker exec -it postgres psql -U postgres -d postgres -c \
  "UPDATE task_service.tasks SET updated_at = NOW() - INTERVAL '5 days' WHERE id = $TASK_ID;"

# 5. Ждать CRON + worker
sleep 40

# 6. Проверить всё
echo "════ Postgres ════"
docker exec -it postgres psql -U postgres -d postgres -c \
  "SELECT id, priority FROM task_service.tasks WHERE id = $TASK_ID;"

echo "════ Notification ════"
docker exec -it postgres psql -U postgres -d postgres -c \
  "SELECT id, notification_type, status, error_message 
   FROM task_notification.notifications WHERE task_id = $TASK_ID ORDER BY id;"

echo "════ Analytics ════"
docker exec -it postgres psql -U postgres -d postgres -c \
  "SELECT event_type, priority FROM analytics_service.task_analytics 
   WHERE title = 'E2E escalation v4' ORDER BY processed_at;"

echo "════ MailHog ════"
curl -s http://localhost:8025/api/v2/messages | python3 -c "
import sys, json
items = json.load(sys.stdin).get('items', [])
for m in items[:5]:
    print(m['Content']['Headers'].get('Subject', [''])[0])
"

echo "════ task-worker ════"
docker logs task-worker --tail 30 | grep -iE "escalat|publish|error"
